// The side bands, the top band, the fill share, and the layout boxes of the three shipped
// compositions, with the exact geometry of the composed picture (186 x 180 mm at 300 ppi).
// Run again: cp this file to /home/daniebo/Desktop/Workbooks/dashboard/c2-composed-gaps.mts
//            cd /home/daniebo/Desktop/Workbooks/dashboard && npx tsx c2-composed-gaps.mts
//            (delete the copy from the dashboard folder afterwards)
import sharp from 'sharp';
import {
    STORYTIME_COVER_FILL_SHARE,
    STORYTIME_COVER_GROUND_TINTS,
    STORYTIME_COVER_SIDE_BAND_MM,
    loadStorytimeCoverBooks,
    storytimeCoverMmToPx,
} from './lib/storytime/print/covers';
import {
    checkStorytimeCoverFigureAspect,
    checkStorytimeCoverFigureLayout,
    checkStorytimeCoverPicture,
    composeStorytimeCoverPicture,
    storytimeCoverFigureLayout,
    storytimeCoverPicturePath,
} from './lib/storytime/print/cover-picture';

const size = { width: storytimeCoverMmToPx(186), height: storytimeCoverMmToPx(180) };
const band = storytimeCoverMmToPx(8);
const side = storytimeCoverMmToPx(STORYTIME_COVER_SIDE_BAND_MM);
const rows = storytimeCoverMmToPx(6);
const MM = (px: number) => (px / 300 * 25.4).toFixed(2);
const data = loadStorytimeCoverBooks();
let failures = 0;
const report = (label: string, ok: boolean, detail: string) => {
    if (!ok) failures++;
    console.log(`${ok ? 'PASS' : 'FAIL'} ${label}: ${detail}`);
};

for (const [level, unit] of [[1, 1], [2, 1], [4, 4]] as const) {
    const book = data.books.find((row) => row.level === level && row.unit === unit)!;
    const files = book.pictureRecipe.map((figure) => storytimeCoverPicturePath(figure.file));
    const tint = STORYTIME_COVER_GROUND_TINTS[level];
    const png = await composeStorytimeCoverPicture(files, size, tint, book.title);
    const raw = await sharp(png).raw().toBuffer({ resolveWithObject: true });
    const ground = [1, 3, 5].map((index) => Number.parseInt(tint.slice(index, index + 2), 16));
    const isFigure = (x: number, y: number) => {
        const index = (y * raw.info.width + x) * raw.info.channels;
        return [0, 1, 2].some((channel) => Math.abs(raw.data[index + channel] - ground[channel]) > 24);
    };
    let figure = 0;
    for (let y = 0; y < raw.info.height; y++) for (let x = 0; x < raw.info.width; x++) if (isFigure(x, y)) figure++;
    let left = raw.info.width;
    let right = -1;
    for (let x = 0; x < raw.info.width; x++) {
        for (let y = rows; y < raw.info.height - rows; y++) {
            if (isFigure(x, y)) {
                left = Math.min(left, x);
                right = Math.max(right, x);
                break;
            }
        }
    }
    let top = raw.info.height;
    for (let y = 0; y < raw.info.height; y++) {
        for (let x = rows; x < raw.info.width - rows; x++) if (isFigure(x, y)) { top = Math.min(top, y); break; }
    }
    const boxes = await storytimeCoverFigureLayout(files, size);
    const layoutErrors = checkStorytimeCoverFigureLayout(boxes, size, band);
    const pictureErrors = checkStorytimeCoverPicture(png, tint, STORYTIME_COVER_FILL_SHARE);
    console.log(`\n== Level ${level} Unit ${unit} ==`);
    console.log(`boxes ${JSON.stringify(boxes.map((box) => `${box.name} ${box.left}+${box.width},${box.top}+${box.height}`))}`);
    report('composed fill share', figure / (raw.info.width * raw.info.height) >= STORYTIME_COVER_FILL_SHARE, `${(figure / (raw.info.width * raw.info.height)).toFixed(4)} (threshold ${STORYTIME_COVER_FILL_SHARE})`);
    report('left side gap', left >= side, `${MM(left)} mm (${left} px); the side band is ${STORYTIME_COVER_SIDE_BAND_MM} mm; the test scans its ${side} px`);
    report('right side gap', raw.info.width - 1 - right >= side, `${MM(raw.info.width - 1 - right)} mm (${raw.info.width - 1 - right} px)`);
    report('top gap', top >= band, `${MM(top)} mm (${top} px); the head band is 8 mm = ${band} px`);
    const aspectErrors = checkStorytimeCoverFigureAspect(boxes, files, 0.02);
    report('aspect check', aspectErrors.length === 0, aspectErrors.length ? aspectErrors.join(' | ') : 'no error');
    report('layout check', layoutErrors.length === 0, layoutErrors.length ? layoutErrors.join(' | ') : 'no error');
    report('picture check', pictureErrors.length === 0, pictureErrors.length ? pictureErrors.join(' | ') : 'no error');
}
console.log(`\ncomposed side and top bands: ${failures === 0 ? 'all pass' : `${failures} fail`}`);
process.exit(failures === 0 ? 0 : 1);
